A complete data table solution with advanced filtering, sorting, pagination, and
URL state management. This is a barrel export that re-exports all data-table
sub-components from `@/components/edust-kit/*`.

## Usage

```tsx
import { DataTable, DataTableToolbar } from "@/components/edust-kit";
import { useDataTable } from "@/components/edust-kit/use-data-table";
```

## Async Faceted Filter

The `DataTableAsyncFacetedFilter` component provides a faceted filter that
fetches options asynchronously from an API. It supports debounced search,
loading states, error handling, and displaying selected items that are not
currently in the fetched results.

### Props

| Prop | Type | Description |
|------|------|-------------|
| `column` | `Column<TData, TValue>` | The TanStack Table column to filter |
| `title` | `string` | Display label for the filter button |
| `asyncOptions` | `AsyncColumnOptions` | Configuration for async data fetching |

### AsyncColumnOptions

```typescript
interface AsyncColumnOptions {
  useDataHook: (params: Record<string, unknown>) => {
    data?: { data?: { items?: unknown[] } };
    isFetching: boolean;
    error: unknown;
  };
  searchParamKey?: string;       // Default: "search"
  getItemLabel: (item: unknown) => string;
  getItemValue?: (item: unknown) => string;  // Default: item.id
  additionalParams?: Record<string, unknown>;
  debounceDelay?: number;        // Default: 300
}
```

### Usage

```tsx
import { DataTableAsyncFacetedFilter } from "@/components/edust-kit/data-table-async-faceted-filter";
import { useQuery } from "@tanstack/react-query";

function useUsers(params: Record<string, unknown>) {
  return useQuery({
    queryKey: ["users", params],
    queryFn: () => fetchUsers(params),
  });
}

<DataTableAsyncFacetedFilter
  column={table.getColumn("userId")}
  title="User"
  asyncOptions={{
    useDataHook: useUsers,
    getItemLabel: (item) => item.name,
    getItemValue: (item) => item.id,
    searchParamKey: "search",
    debounceDelay: 300,
  }}
/>
```
