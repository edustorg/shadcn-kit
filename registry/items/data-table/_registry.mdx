A complete data table solution with advanced filtering, sorting, pagination, and
URL state management. This is a barrel export that re-exports all data-table
sub-components from `@/components/edust-kit/data-table/*`.

## Usage

```tsx
import { DataTable } from "@/components/edust-kit/data-table/data-table"
import { DataTableToolbar } from "@/components/edust-kit/data-table/data-table-toolbar"
import { useDataTable } from "@/components/edust-kit/data-table/hooks/use-data-table"
```

## Async Faceted Filter

The `DataTableAsyncFacetedFilter` component provides a faceted filter that
fetches options asynchronously from an API. It supports debounced search,
loading states, error handling, and displaying selected items that are not
currently in the fetched results.

### Props

| Prop           | Type                    | Description                           |
| -------------- | ----------------------- | ------------------------------------- |
| `column`       | `Column<TData, TValue>` | The TanStack Table column to filter   |
| `title`        | `string`                | Display label for the filter button   |
| `asyncOptions` | `AsyncColumnOptions`    | Configuration for async data fetching |

### AsyncColumnOptions

```typescript
interface AsyncColumnOptions {
  useDataHook: (params: Record<string, unknown>) => {
    data?: { data?: { items?: unknown[] } }
    isFetching: boolean
    error: unknown
  }
  searchParamKey?: string // Default: "search"
  getItemLabel: (item: unknown) => string
  getItemValue?: (item: unknown) => string // Default: item.id
  additionalParams?: Record<string, unknown>
  debounceDelay?: number // Default: 300
}
```

### Usage

```tsx
import { DataTableAsyncFacetedFilter } from "@/components/edust-kit/data-table/data-table-async-faceted-filter"
import { useQuery } from "@tanstack/react-query"

function useUsers(params: Record<string, unknown>) {
  return useQuery({
    queryKey: ["users", params],
    queryFn: () => fetchUsers(params),
  })
}

;<DataTableAsyncFacetedFilter
  column={table.getColumn("userId")}
  title="User"
  asyncOptions={{
    useDataHook: useUsers,
    getItemLabel: (item) => item.name,
    getItemValue: (item) => item.id,
    searchParamKey: "search",
    debounceDelay: 300,
  }}
/>
```

## Action Bar

A floating toolbar that appears when rows are selected in a data table. Renders
as a portal at the bottom of the viewport.

### Props

| Prop             | Type                           | Description                                |
| ---------------- | ------------------------------ | ------------------------------------------ |
| `open`           | `boolean`                      | Whether the action bar is visible          |
| `onOpenChange`   | `(open: boolean) => void`      | Callback when visibility changes           |
| `side`           | `"top" \| "bottom"`            | Which side to anchor to (default: bottom)  |
| `align`          | `"start" \| "center" \| "end"` | Horizontal alignment (default: center)     |
| `sideOffset`     | `number`                       | Distance from edge in px (default: 16)     |
| `orientation`    | `"horizontal" \| "vertical"`   | Layout direction (default: horizontal)     |
| `loop`           | `boolean`                      | Arrow key navigation wraps (default: true) |

### Sub-components

- `ActionBarSelection` — shows selected row count with a close button
- `ActionBarGroup` — groups related action items with roving tabindex
- `ActionBarItem` — individual action button (wraps Button)
- `ActionBarClose` — dismisses the action bar
- `ActionBarSeparator` — visual divider between items

### Usage

```tsx
import {
  ActionBar,
  ActionBarClose,
  ActionBarGroup,
  ActionBarItem,
  ActionBarSelection,
  ActionBarSeparator,
} from "@/components/edust-kit/data-table/action-bar"
import { CheckCircle2, Download, X } from "lucide-react"

function TableActionBar({ table }) {
  const rows = table.getFilteredSelectedRowModel().rows

  const onOpenChange = React.useCallback(
    (open: boolean) => {
      if (!open) table.toggleAllRowsSelected(false)
    },
    [table],
  )

  return (
    <ActionBar open={rows.length > 0} onOpenChange={onOpenChange}>
      <ActionBarSelection>
        <span className="font-medium">{rows.length}</span>
        <span>selected</span>
        <ActionBarSeparator />
        <ActionBarClose>
          <X />
        </ActionBarClose>
      </ActionBarSelection>
      <ActionBarSeparator />
      <ActionBarGroup>
        <ActionBarItem>
          <CheckCircle2 />
          Status
        </ActionBarItem>
        <ActionBarItem>
          <Download />
          Export
        </ActionBarItem>
      </ActionBarGroup>
    </ActionBar>
  )
}
```

Pass it to `DataTable` via the `actionBar` prop:

```tsx
<DataTable table={table} actionBar={<TableActionBar table={table} />} />
```

## DataTable

The `DataTable` component renders the TanStack Table instance with headers, rows,
static pinning styles, pagination and an optional action bar. When `isFetching`
is `true` it renders skeleton rows instead of data.

### Props

| Prop         | Type                                         | Description                                                        |
| ------------ | -------------------------------------------- | ------------------------------------------------------------------ |
| `table`      | `TanstackTable<TData>`                       | The TanStack Table instance                                        |
| `isFetching` | `boolean`                                    | When `true`, renders skeleton loading rows                         |
| `skeleton`   | `{ columnCount?, rowCount?, cellWidths?, shrinkZero? }` | Skeleton shape overrides (defaults to all leaf columns / 10 rows) |
| `actionBar`  | `React.ReactNode`                            | Rendered when rows are selected                                    |

### Usage

```tsx
import { DataTable } from "@/components/edust-kit/data-table/data-table"

<DataTable table={table} isFetching={isFetching} actionBar={<TableActionBar table={table} />}>
  <DataTableToolbar table={table} />
</DataTable>
```
