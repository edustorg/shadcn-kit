A complete data table solution with advanced filtering, sorting, pagination, and
URL state management. This is a barrel export that re-exports all data-table
sub-components from `@/components/edust-kit/data-table/*`.

## Usage

```tsx
import { DataTable } from "@/components/edust-kit/data-table/data-table"
import { DataTableToolbar } from "@/components/edust-kit/data-table/data-table-toolbar"
import { useDataTable } from "@/components/edust-kit/data-table/hooks/use-data-table"
```

## Async Faceted Filter

The `DataTableAsyncFacetedFilter` component provides a faceted filter that
fetches options asynchronously from an API. It supports debounced search,
loading states, error handling, and displaying selected items that are not
currently in the fetched results.

### Props

| Prop           | Type                    | Description                           |
| -------------- | ----------------------- | ------------------------------------- |
| `column`       | `Column<TData, TValue>` | The TanStack Table column to filter   |
| `title`        | `string`                | Display label for the filter button   |
| `asyncOptions` | `AsyncColumnOptions`    | Configuration for async data fetching |

### AsyncColumnOptions

```typescript
interface AsyncColumnOptions {
  useDataHook: (params: Record<string, unknown>) => {
    data?: { data?: { items?: unknown[] } }
    isFetching: boolean
    error: unknown
  }
  searchParamKey?: string // Default: "search"
  getItemLabel: (item: unknown) => string
  getItemValue?: (item: unknown) => string // Default: item.id
  additionalParams?: Record<string, unknown>
  debounceDelay?: number // Default: 300
}
```

### Usage

```tsx
import { DataTableAsyncFacetedFilter } from "@/components/edust-kit/data-table/data-table-async-faceted-filter"
import { useQuery } from "@tanstack/react-query"

function useUsers(params: Record<string, unknown>) {
  return useQuery({
    queryKey: ["users", params],
    queryFn: () => fetchUsers(params),
  })
}

;<DataTableAsyncFacetedFilter
  column={table.getColumn("userId")}
  title="User"
  asyncOptions={{
    useDataHook: useUsers,
    getItemLabel: (item) => item.name,
    getItemValue: (item) => item.id,
    searchParamKey: "search",
    debounceDelay: 300,
  }}
/>
```

## Action Bar

A floating toolbar that appears when rows are selected in a data table. Renders
as a portal at the bottom of the viewport.

### Props

| Prop             | Type                           | Description                                |
| ---------------- | ------------------------------ | ------------------------------------------ |
| `open`           | `boolean`                      | Whether the action bar is visible          |
| `onOpenChange`   | `(open: boolean) => void`      | Callback when visibility changes           |
| `side`           | `"top" \| "bottom"`            | Which side to anchor to (default: bottom)  |
| `align`          | `"start" \| "center" \| "end"` | Horizontal alignment (default: center)     |
| `sideOffset`     | `number`                       | Distance from edge in px (default: 16)     |
| `orientation`    | `"horizontal" \| "vertical"`   | Layout direction (default: horizontal)     |
| `loop`           | `boolean`                      | Arrow key navigation wraps (default: true) |

### Sub-components

- `ActionBarSelection` — shows selected row count with a close button
- `ActionBarGroup` — groups related action items with roving tabindex
- `ActionBarItem` — individual action button (wraps Button)
- `ActionBarClose` — dismisses the action bar
- `ActionBarSeparator` — visual divider between items

### Usage

```tsx
import {
  ActionBar,
  ActionBarClose,
  ActionBarGroup,
  ActionBarItem,
  ActionBarSelection,
  ActionBarSeparator,
} from "@/components/edust-kit/data-table/action-bar"
import { CheckCircle2, Download, X } from "lucide-react"

function TableActionBar({ table }) {
  const rows = table.getFilteredSelectedRowModel().rows

  const onOpenChange = React.useCallback(
    (open: boolean) => {
      if (!open) table.toggleAllRowsSelected(false)
    },
    [table],
  )

  return (
    <ActionBar open={rows.length > 0} onOpenChange={onOpenChange}>
      <ActionBarSelection>
        <span className="font-medium">{rows.length}</span>
        <span>selected</span>
        <ActionBarSeparator />
        <ActionBarClose>
          <X />
        </ActionBarClose>
      </ActionBarSelection>
      <ActionBarSeparator />
      <ActionBarGroup>
        <ActionBarItem>
          <CheckCircle2 />
          Status
        </ActionBarItem>
        <ActionBarItem>
          <Download />
          Export
        </ActionBarItem>
      </ActionBarGroup>
    </ActionBar>
  )
}
```

Pass it to `DataTable` via the `actionBar` prop:

```tsx
<DataTable table={table} actionBar={<TableActionBar table={table} />} />
```

## DataTable

The `DataTable` component renders the TanStack Table instance with headers, rows,
static pinning styles, pagination and an optional action bar. When `isFetching`
is `true` it renders skeleton rows instead of data.

### Props

| Prop         | Type                                         | Description                                                        |
| ------------ | -------------------------------------------- | ------------------------------------------------------------------ |
| `table`      | `TanstackTable<TData>`                       | The TanStack Table instance                                        |
| `isFetching` | `boolean`                                    | When `true`, renders skeleton loading rows                         |
| `skeleton`   | `{ columnCount?, rowCount?, cellWidths?, shrinkZero? }` | Skeleton shape overrides (defaults to all leaf columns / 10 rows) |
| `actionBar`  | `React.ReactNode`                            | Rendered when rows are selected                                    |

### Usage

```tsx
import { DataTable } from "@/components/edust-kit/data-table/data-table"

<DataTable table={table} isFetching={isFetching} actionBar={<TableActionBar table={table} />}>
  <DataTableToolbar table={table} />
</DataTable>
```

## Query Params

Utilities for mapping browser URL state into API query params. They cover both
supported browser URL formats for multi-value params.

```text
?assignee=id1,id2            ?assignee=id1&assignee=id2
        ↓                             ↓
getArrayParam(params, "assignee")     getArrayParam(params, "assignee")
        ↓                             ↓
["id1", "id2"]                        ["id1", "id2"]
```

Return a real array so your HTTP client (axios `params`, qs `indices`) serializes
it as `filter_by_assignee_ids[0]=id1&filter_by_assignee_ids[1]=id2` — never a
comma-joined single string.

### Functions

| Function                                        | Description                                          |
| ----------------------------------------------- | ---------------------------------------------------- |
| `getArrayParam(params, key, delimiter?)`        | `string[]` from comma-separated or repeated params   |
| `getBooleanParam(params, key)`                  | `boolean` from `"true"` / `"false"`                  |
| `getNumberParam(params, key, fallback?)`        | `number` from a single value                         |
| `getRangeParam(params, key, delimiter?)`        | `{ from?, to? }` from `"from,to"`                    |

### Usage

```tsx
import { useSearchParams } from "next/navigation"
import { useQuery } from "@tanstack/react-query"
import {
  getArrayParam,
  getBooleanParam,
  getNumberParam,
  getRangeParam,
} from "@/components/edust-kit/data-table/lib/query-params"

function useTasksQuery(options?: { userId?: string }) {
  const params = useSearchParams()

  const query = {
    search: params.get("title") || undefined,
    status: getArrayParam(params, "status"),
    priority: getArrayParam(params, "priority"),
    filter_by_assignee_ids: options?.userId
      ? [options.userId]
      : getArrayParam(params, "assignee"),
    filter_by_created_by_ids: getArrayParam(params, "createdBy"),
    filter_by_label_ids: getArrayParam(params, "labelId"),
    range_by_due_date: getRangeParam(params, "dueDate"),
    filter_by_unassigned: getBooleanParam(params, "filter_by_unassigned"),
    page: getNumberParam(params, "page", 1),
    limit: getNumberParam(params, "limit", 10),
  }

  return query
}

function useTasks() {
  const query = useTasksQuery()

  return useQuery({
    queryKey: ["tasks", query],
    queryFn: async () => {
      const res = await axios.get(baseUrl, { params: query })
      return res.data
    },
  })
}
```

## Date Formatting

The `lib/date-format.ts` provides utilities for formatting dates in data tables using
[date-fns](https://date-fns.org/). It includes 9 format variants and 2 helper functions.

### Variants

| Variant          | Description                            | Example Output                          |
| ---------------- | -------------------------------------- | --------------------------------------- |
| `relative`       | Full relative with time                | "Today at 10:35 AM" / "Jun 16, 2025 at 10:35 AM" |
| `relative-short` | Compact relative                       | "Today 10:35 AM" / "Wed 10:35 AM" / "Jun 16, 2025" |
| `short`          | Date only                              | "Jun 16, 2026"                          |
| `medium`         | Date + time (default)                  | "Jun 16, 2026 10:35 AM"                 |
| `long`           | Long month + date + time               | "June 16, 2026 10:35 AM"                |
| `full`           | Full weekday + month + date + time     | "Tuesday, June 16, 2026 at 10:35 AM"    |
| `time-only`      | Time only                              | "10:35 AM"                              |
| `date-only`      | Date only (short month)                | "Jun 16, 2026"                          |
| `iso`            | ISO 8601                               | "2026-06-16T10:35:00+00:00"             |

### Functions

```typescript
import {
  formatDate,
  formatRelativeDate,
  formatRelativeDateShort,
  type DateFormatVariant,
} from "@/components/edust-kit/data-table/lib/date-format"
```

| Function                      | Description                                    |
| ----------------------------- | ---------------------------------------------- |
| `formatDate(date, variant?)`  | Main formatter with variant support            |
| `formatRelativeDate(date)`    | Shortcut for `formatDate(date, "relative")`    |
| `formatRelativeDateShort(date)` | Shortcut for `formatDate(date, "relative-short")` |

### Usage in DataTable Columns

```tsx
import { formatDate } from "@/components/edust-kit/data-table/lib/date-format"

const columns = [
  {
    accessorKey: "createdAt",
    header: "Created",
    cell: ({ row }) => formatDate(row.getValue("createdAt"), "relative-short"),
  },
  {
    accessorKey: "updatedAt",
    header: "Updated",
    cell: ({ row }) => formatDate(row.getValue("updatedAt"), "relative"),
  },
]
```

### Input Handling

Accepts `Date`, ISO string, timestamp (number), `null`, or `undefined`. Returns `"-"` for invalid/missing dates.

```tsx
formatDate(new Date("2026-06-16T10:35:00Z"), "relative")     // "Today at 10:35 AM"
formatDate("2026-06-15T10:35:00Z", "relative-short")        // "Yesterday 10:35 AM"
formatDate(1718620500000, "medium")                          // "Jun 16, 2026 10:35 AM"
formatDate(null, "short")                                    // "-"
```
